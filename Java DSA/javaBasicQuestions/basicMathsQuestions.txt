Q1 - sum Of Two Numbers

import java.util.Scanner;

public class Solution{
    public static int  sumOfTwoNumbers() {
     
      Scanner sc = new Scanner(System.in);

      int a = sc.nextInt();
      int b = sc.nextInt();
      int sum = a + b;
    
        // add both integer and return their sum 

        return sum;
    }
}
